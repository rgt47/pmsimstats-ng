% Three-part paragraph scaffolding for manuscript revision.
\usepackage{xcolor}

\newenvironment{bullets}{%
  \begin{quote}\small\sffamily\color[gray]{0.30}}{%
  \end{quote}}

\newenvironment{rgt}{%
  \begin{quote}\normalsize\color[RGB]{20,60,160}}{%
  \end{quote}}

\newenvironment{orig}{%
  \begin{quote}\small\itshape\color[gray]{0.55}}{%
  \end{quote}}

\newenvironment{claudecode}{\begin{orig}}{\end{orig}}
