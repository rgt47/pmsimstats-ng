% Statistics in Medicine submission preamble.
% Shared across analysis/report/<paper>/report.Rmd files via
%   includes:
%     in_header: ../sim-preamble.tex
% in the YAML output block.

% Continuous line numbering for editorial review. The 'mathlines'
% option extends numbering through display-math environments
% (equation, align, ...), which SIM editorial workflows expect.
\usepackage[mathlines]{lineno}
\linenumbers
\usepackage{amsmath}

% Spacing: linestretch is set in YAML (1.5 line spacing).
\usepackage{setspace}

% Page footer with source path and render timestamp.
% Each paper's report.Rmd should include a setup chunk that
% renews \rmdsource and \rendertimestamp at knit time. Defaults
% below are placeholders shown when the chunk is missing.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\newcommand{\rmdsource}{Source: unspecified}
\newcommand{\rendertimestamp}{Rendered: \today}
\fancyfoot[L]{\footnotesize\itshape \rmdsource}
\fancyfoot[C]{\footnotesize\itshape \rendertimestamp}
\fancyfoot[R]{\footnotesize\thepage}
