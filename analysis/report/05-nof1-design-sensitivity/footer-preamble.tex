% Page footer with source path and render timestamp.
% The report.Rmd footer-setup chunk renews \rmdsource and
% \rendertimestamp at knit time. The \providecommand defaults
% below are placeholders shown when the chunk is missing.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\providecommand{\rmdsource}{Source: unspecified}
\providecommand{\rendertimestamp}{Rendered: \today}
\fancyfoot[L]{\footnotesize\itshape \rmdsource}
\fancyfoot[C]{\footnotesize\itshape \rendertimestamp}
\fancyfoot[R]{\footnotesize\thepage}
